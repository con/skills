# Pinned tooling for the eval harness.  P.2/P.3 apply to our own tests too, so
# this is the spec the harness host installs from:
#
#   uv tool install datalad --with git-annex --with-executables-from git-annex
#   uv pip install -r evals/stamped/requirements.txt
#
# datalad 1.7.1 carries the concurrent-run fix (datalad#7899, #7901); do not
# relax that lower bound while outer-loop runs may overlap.
datalad==1.7.1
datalad-container==1.2.6
# The plain 10.20260901 release ships only macOS and Windows wheels; .post1 is
# the one that carries manylinux_2_34_x86_64 (and aarch64). Pinning the release
# without the .post1 makes this spec uninstallable on Linux — which is the
# P.2/P.3 defect the probes hunt, in our own environment file.
git-annex==10.20260901.post1
con-duct==0.16.0
PyYAML==6.0.2
pytest==8.3.5
# The aggregation layer only: one eval case per mutant, one analysis test per
# criterion (docs/stamped/eval-platforms.md, recommendation 2).
pytest-evals==0.3.4
